```rust
extern crate proc_macro;
use proc_macro::TokenStream;
use quote::quote;
use std::collections::BTreeMap;
use syn::{
    parse::{Parse, ParseStream},
    parse_macro_input, Expr, LitStr, Token,
};

// Represents a key-action pair: "ga" => expr
struct KeyBinding {
    key: LitStr,
    action: Expr,
}

impl Parse for KeyBinding {
    fn parse(input: ParseStream) -> syn::Result<Self> {
        let key: LitStr = input.parse()?;
        input.parse::<Token![:]>()?;
        let action: Expr = input.parse()?;
        Ok(KeyBinding { key, action })
    }
}

struct KeyMapInput {
    bindings: Vec<KeyBinding>,
}

impl Parse for KeyMapInput {
    fn parse(input: ParseStream) -> syn::Result<Self> {
        let bindings = input.parse_terminated(KeyBinding::parse, Token![,])?;
        Ok(KeyMapInput {
            bindings: bindings.into_iter().collect(),
        })
    }
}

// Compile-time Trie Node
#[derive(Default)]
struct TrieNode {
    action: Option<Expr>,
    children: BTreeMap<char, TrieNode>,
}

impl TrieNode {
    fn insert(&mut self, key: &str, action: Expr, span: proc_macro2::Span) -> syn::Result<()> {
        let mut current = self;
        let chars: Vec<char> = key.chars().collect();

        for (i, &ch) in chars.iter().enumerate() {
            // Conflict Rule 1: A prefix of this key already triggers an action
            if current.action.is_some() {
                return Err(syn::Error::new(
                    span,
                    format!("Key sequence '{}' conflicts with an already defined shorter key sequence.", key),
                ));
            }

            current = current.children.entry(ch).or_default();

            // Conflict Rule 2: This key is a prefix of an existing longer sequence
            if i == chars.len() - 1 && (!current.children.is_empty() || current.action.is_some()) {
                return Err(syn::Error::new(
                    span,
                    format!("Key sequence '{}' cannot be mapped because it is a prefix for composite keys.", key),
                ));
            }
        }

        current.action = Some(action);
        Ok(())
    }
}

#[proc_macro]
pub fn keymap(input: TokenStream) -> TokenStream {
    let KeyMapInput { bindings } = parse_macro_input!(input as KeyMapInput);
    let mut root = TrieNode::default();

    for binding in bindings {
        let key_str = binding.key.value();
        if let Err(err) = root.insert(&key_str, binding.action, binding.key.span()) {
            return err.into_compile_error().into();
        }
    }

    let match_tree = generate_match_level(&root);

    let expanded = quote! {
        // Enclosure expecting a key-fetching closure or iterator
        move |mut get_next_key: impl FnMut() -> Option<char>| {
            #match_tree
        }
    };

    TokenStream::from(expanded)
}

// Recursively builds nested `match` statements
fn generate_match_level(node: &TrieNode) -> proc_macro2::TokenStream {
    if let Some(ref action) = node.action {
        return quote! { Some(#action) };
    }

    let arms = node.children.iter().map(|(&ch, child_node)| {
        let child_match = generate_match_level(child_node);
        quote! {
            #ch => #child_match,
        }
    });

    quote! {
        if let Some(key) = get_next_key() {
            match key {
                #(#arms)*
                _ => None,
            }
        } else {
            None
        }
    }
}
```

For the editor core API, I’d make the keystrokes dispatch into a relatively small set of semantic operations rather than having one API function per key. The key handler can translate keys/motions into these operations.

Visual mode

Key	Core operation	Intended behavior
h/j/k/l + motions	expand_selection(motion)	Expand the block selection according to the motion.
y	yank_selection()	Copy the selected block.

The important distinction is that visual movement should not itself mutate the table; it only changes the Selection.

Move mode

Move mode should intentionally have an extremely small API surface:

Key	Core operation	Intended behavior
h	move_selection_left(count)	Move the selected range left.
j	move_selection_down(count)	Move the selected range down.
k	move_selection_up(count)	Move the selected range up.
l	move_selection_right(count)	Move the selected range right.
motions	move_selection(motion)	Move the selected range according to the resolved motion.

No editing commands should be dispatched from move mode.

Command mode
Input	Core operation	Intended behavior
:q	quit()	Exit the application/document.
:w	save()	Save without quitting.

I'd make command mode ultimately call the same save()/quit() operations as the ordinary keybindings rather than implementing separate behavior.

One particularly useful architectural operation

I'd also have the keybinding layer expose:

dispatch_key(key, count, mode)

Its job is not to modify the document directly. It resolves:

raw keystrokes
    ↓
key sequence / count parser
    ↓
semantic operation + arguments
    ↓
core operation
    ↓
document state mutation


For example:

3j
→ dispatch_key(...)
→ move_down(count=3)


and:

d2j
→ parse operator + motion
→ delete_range(motion=down, count=2)


This keeps the core independent of Vim's particular keybindings. You can later change j to another binding, add alternative bindings, or provide a GUI command without changing the underlying table operations.
